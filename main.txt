class Car {
  final String id;
  final String brand;
  final String model;
  final String generation;
  final BodyType bodyType;
  final String color;
  final Rarity rarity;
  final int baseValue;
  int currentValue;
  int speed;
  int handling;
  int durability;
  final int maxSpeed;
  final int maxHandling;
  final int maxDurability;
  int level;
  int experience;
  int damage;
  String? photoPath;
  final DateTime discoveredAt;
  CarStatus status;
  int? garageSlot;

  Car({
    required this.id,
    required this.brand,
    required this.model,
    required this.generation,
    required this.bodyType,
    required this.color,
    required this.rarity,
    required this.baseValue,
    required this.currentValue,
    required this.speed,
    required this.handling,
    required this.durability,
    required this.maxSpeed,
    required this.maxHandling,
    required this.maxDurability,
    required this.level,
    required this.experience,
    required this.damage,
    required this.discoveredAt,
    required this.status,
    this.photoPath,
    this.garageSlot,
  });

  String get title => '$brand $model';
  bool get isInGarage => status == CarStatus.garaged;
  bool get isUsable => status != CarStatus.stolen && status != CarStatus.inChase;

  double get condition => 1 - damage.clamp(0, 100) / 100.0;

  int get repairCost {
    // 25% от стоимости восстановления полного повреждения.
    return ((currentValue * (damage / 100)) * 0.25).round();
  }

  Map<String, dynamic> toJson() => {
        'id': id,
        'brand': brand,
        'model': model,
        'generation': generation,
        'bodyType': bodyType.name,
        'color': color,
        'rarity': rarity.name,
        'baseValue': baseValue,
        'currentValue': currentValue,
        'speed': speed,
        'handling': handling,
        'durability': durability,
        'maxSpeed': maxSpeed,
        'maxHandling': maxHandling,
        'maxDurability': maxDurability,
        'level': level,
        'experience': experience,
        'damage': damage,
        'photoPath': photoPath,
        'discoveredAt': discoveredAt.toIso8601String(),
        'status': status.name,
        'garageSlot': garageSlot,
  };

  factory Car.fromJson(Map<String, dynamic> j) => Car(
        id: j['id'],
        brand: j['brand'],
        model: j['model'],
        generation: j['generation'],
        bodyType: BodyType.values.byName(j['bodyType']),
        color: j['color'],
        rarity: Rarity.values.byName(j['rarity']),
        baseValue: j['baseValue'],
        currentValue: j['currentValue'],
        speed: j['speed'],
        handling: j['handling'],
        durability: j['durability'],
        maxSpeed: j['maxSpeed'],
        maxHandling: j['maxHandling'],
        maxDurability: j['maxDurability'],
        level: j['level'],
        experience: j['experience'],
        damage: j['damage'],
        photoPath: j['photoPath'],
        discoveredAt: DateTime.parse(j['discoveredAt']),
        status: CarStatus.values.byName(j['status']),
        garageSlot: j['garageSlot'],
      );
}

class RarityService {
  static const catalog = <VehicleCatalogEntry>[

      collectability: 72,
    ),
    VehicleCatalogEntry(

      model: '911',
      rarityScore: 74,
      prestige: 90,
      baseValue: 140000,
      popularity: 55,
      sportiness: 95,
      collectability: 92,
    ),
    VehicleCatalogEntry(
      brand: 'Nissan',
      model: 'GT-R',
      rarityScore: 76,
      prestige: 80,
      baseValue: 120000,
      popularity: 45,
      sportiness: 98,
      collectability: 94,
    ),
    VehicleCatalogEntry(
      brand: 'Ford',
      model: 'Mustang',
      rarityScore: 45,
      prestige: 55,
      baseValue: 50000,
      popularity: 78,
      sportiness: 84,
      collectability: 70,
    ),
    VehicleCatalogEntry(
      brand: 'Lamborghini',
      model: 'Huracán',
      rarityScore: 94,
      prestige: 100,
      baseValue: 260000,
      popularity: 18,
      sportiness: 100,
      collectability: 100,
    ),
    VehicleCatalogEntry(
      brand: 'Tesla',
      model: 'Model S',
      rarityScore: 40,
      prestige: 70,
      baseValue: 90000,
      popularity: 82,
      sportiness: 78,
      collectability: 62,
    ),
  ];

  Rarity calculate({
    required String brand,
    required String model,
    required String color,
  }) {
    final c = catalog.firstWhere(
      (e) => e.brand == brand && e.model == model,
      orElse: () => const VehicleCatalogEntry(
        brand: '',
        model: '',
        rarityScore: 25,
        prestige: 30,
        baseValue: 25000,
        popularity: 80,
        sportiness: 40,
        collectability: 25,
      ),
    );

    final unusualColor = _unusualColorScore(color);
    final popularityInverse = 100 - c.popularity;

    final score = (
          c.rarityScore * 0.22 +
          c.prestige * 0.18 +
          _valueScore(c.baseValue) * 0.15 +
          popularityInverse * 0.12 +
          unusualColor * 0.08 +
          c.sportiness * 0.10 +
          c.collectability * 0.15
        )
        .round();

    if (score >= 82) return Rarity.legendary;
    if (score >= 66) return Rarity.epic;
    if (score >= 49) return Rarity.rare;
    if (score >= 31) return Rarity.uncommon;
    return Rarity.common;
  }

  int baseValueFor(String brand, String model) {
    final c = catalog.where((e) => e.brand == brand && e.model == model);
    return c.isEmpty ? 25000 : c.first.baseValue;
  }

  int _valueScore(int value) {
    final normalized = (log(max(1, value)) / log(500000)) * 100;
    return normalized.clamp(0, 100).round();
  }

  int _unusualColorScore(String color) {
    const common = {'Белый', 'Чёрный', 'Серый', 'Серебристый'};
    return common.contains(color) ? 25 : 75;
  }
}

class ExperienceService {
  int requiredForLevel(int level) => 100 + ((level - 1) * 75) + ((level - 1) * (level - 1) * 15);

  int levelFromTotalXp(int totalXp) {
    var level = 1;
    var xp = totalXp;
    while (xp >= requiredForLevel(level)) {
      xp -= requiredForLevel(level);
      level++;
    }
    return level;
  }

  int xpForDiscovery(Rarity rarity) => switch (rarity) {
        Rarity.common => 25,
        Rarity.uncommon => 45,
        Rarity.rare => 80,
        Rarity.epic => 150,
        Rarity.legendary => 300,
      };

  int xpForRace({required bool win}) => win ? 120 : 45;
  int xpForChase() => 160;
}

class EconomyService {
  int discoveryReward(Rarity rarity) => switch (rarity) {
        Rarity.common => 500,
        Rarity.uncommon => 900,
        Rarity.rare => 1500,
        Rarity.epic => 3000,
        Rarity.legendary => 7000,
      };

  int raceReward({required bool win}) => win ? 1500 : 500;
  int chaseReward() => 1800;

  int garagePrice(int newSlot) => 5000 + (newSlot - 5) * 3500;

  int upgradeCost(Car car, UpgradeType type) {
    final current = switch (type) {
      UpgradeType.engine => car.speed,
      UpgradeType.handling => car.handling,
      UpgradeType.body => car.durability,
    };
    return max(250, (car.currentValue * 0.035 * car.level * (1 + current / 100)).round());
  }

  int salePrice(Car car) => (car.currentValue * 0.5).round();
}

class GarageService {
  int slots = 5;

  bool canStore(Car car, List<Car> cars) =>
      cars.where((c) => c.garageSlot != null).length < slots;

  void store(Car car, List<Car> cars) {
    if (!canStore(car, cars)) {
      throw StateError('Нет свободного места в гараже');
    }
    final used = cars.map((c) => c.garageSlot).whereType<int>().toSet();
    var slot = 0;
    while (used.contains(slot)) {
      slot++;
    }
    car.garageSlot = slot;
    car.status = CarStatus.garaged;
  }

  void remove(Car car) {
    car.garageSlot = null;
    car.status = CarStatus.outside;
  }
}

class UpgradeService {
  void upgrade(Car car, UpgradeType type) {
    switch (type) {
      case UpgradeType.engine:
        if (car.speed >= car.maxSpeed) throw StateError('Скорость уже максимальна');
        car.speed++;
        break;
      case UpgradeType.handling:
        if (car.handling >= car.maxHandling) throw StateError('Манёвренность уже максимальна');
        car.handling++;
        break;
      case UpgradeType.body:
        if (car.durability >= car.maxDurability) throw StateError('Прочность уже максимальна');
        car.durability++;
        break;
    }
    car.level++;
    car.currentValue = (car.currentValue * 1.012).round();
  }

  void applyDamage(Car car, int amount) {
    final reduction = (amount * (car.durability / max(1, car.maxDurability)) * 0.35).round();
    car.damage = (car.damage + max(1, amount - reduction)).clamp(0, 100);
  }

  void repair(Car car) => car.damage = 0;
}

abstract class RaceNetworkService {
  Future<bool> discoverNearby();
  Future<bool> connect(String peerId);
  Future<bool> requestRace(String peerId, String carId, {bool stake = false});
  Future<void> disconnect();
}

class MockRaceNetworkService implements RaceNetworkService {
  bool connected = false;

  @override
  Future<bool> discoverNearby() async {
    await Future<void>.delayed(const Duration(milliseconds: 400));
    return true;
  }

  @override
  Future<bool> connect(String peerId) async {
    await Future<void>.delayed(const Duration(milliseconds: 400));
    connected = true;
    return true;
  }

  @override
  Future<bool> requestRace(String peerId, String carId, {bool stake = false}) async {
    if (!connected) return false;
    await Future<void>.delayed(const Duration(milliseconds: 250));
    return true;
  }

  @override
  Future<void> disconnect() async => connected = false;
}

class TradeOffer {
  final String transactionId;
  final String offeredCarId;
  final String requestedCarId;
  bool firstConfirmed;
  bool secondConfirmed;

  TradeOffer({
    required this.transactionId,
    required this.offeredCarId,
    required this.requestedCarId,
    this.firstConfirmed = false,
    this.secondConfirmed = false,
  });

  bool get ready => firstConfirmed && secondConfirmed;
}

class TradeService {
  final Set<String> activeTransactions = {};

  bool begin(String txId, String carA, String carB) {
    if (activeTransactions.contains(carA) || activeTransactions.contains(carB)) return false;
    activeTransactions.add(carA);
    activeTransactions.add(carB);
    return true;
  }

  void finish(String txId, String carA, String carB) {
    activeTransactions.remove(carA);
    activeTransactions.remove(carB);
  }
}

class GameState {
  int money;
  int totalXp;
  int garageSlots;
  List<Car> cars;

  GameState({
    required this.money,
    required this.totalXp,
    required this.garageSlots,
    required this.cars,
  });

  Map<String, dynamic> toJson() => {
        'money': money,
        'totalXp': totalXp,
        'garageSlots': garageSlots,
        'cars': cars.map((c) => c.toJson()).toList(),
      };

  factory GameState.fromJson(Map<String, dynamic> j) => GameState(
        money: j['money'] ?? 10000,
        totalXp: j['totalXp'] ?? 0,
        garageSlots: j['garageSlots'] ?? 5,
        cars: ((j['cars'] ?? []) as List)
            .map((e) => Car.fromJson(Map<String, dynamic>.from(e)))
            .toList(),
      );
}

class GameStore {
  final SharedPreferences prefs;
  static const _key = 'speed_sanctuary_state_v1';

  GameState state;

  GameStore._(this.prefs, this.state);

  static Future<GameStore> create() async {
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getString(_key);

    if (raw == null) {
      return GameStore._(
        prefs,
        GameState(money: 10000, totalXp: 0, garageSlots: 5, cars: []),
      );
    }

    try {
      return GameStore._(
        prefs,
        GameState.fromJson(jsonDecode(raw)),
      );
    } catch (_) {
      return GameStore._(
        prefs,
        GameState(money: 10000, totalXp: 0, garageSlots: 5, cars: []),
      );
    }
  }

  Future<void> save() async {
    await prefs.setString(_key, jsonEncode(state.toJson()));
  }
}

class GameController extends ChangeNotifier {
  final GameStore store;
  final VehicleRecognitionService recognition;
  final LicensePlateDetectionService plateDetection;
  final ImagePrivacyService privacy;
  final RarityService rarityService;
  final ExperienceService experienceService;
  final EconomyService economyService;
  final GarageService garageService;
  final UpgradeService upgradeService;
  final MarketService marketService;
  final RaceNetworkService networkService;
  final TradeService tradeService;

  GameController({
    required this.store,
    VehicleRecognitionService? recognition,
    LicensePlateDetectionService? plateDetection,
    ImagePrivacyService? privacy,
    RarityService? rarityService,
    ExperienceService? experienceService,
    EconomyService? economyService,
    GarageService? garageService,
    UpgradeService? upgradeService,
    MarketService? marketService,
    RaceNetworkService? networkService,
    TradeService? tradeService,
  })  : recognition = recognition ?? MockVehicleRecognitionService(),
        plateDetection = plateDetection ?? MockLicensePlateDetectionService(),
        privacy = privacy ?? MockImagePrivacyService(),
        rarityService = rarityService ?? RarityService(),
        experienceService = experienceService ?? ExperienceService(),
        economyService = economyService ?? EconomyService(),
        garageService = garageService ?? GarageService(),
        upgradeService = upgradeService ?? UpgradeService(),
        marketService = marketService ?? MarketService(),
        networkService = networkService ?? MockRaceNetworkService(),
        tradeService = tradeService ?? TradeService() {
    garageService.slots = store.state.garageSlots;
  }

  GameState get state => store.state;

  int get playerLevel => experienceService.levelFromTotalXp(state.totalXp);

  int get xpIntoLevel {
    var xp = state.totalXp;
    var level = 1;
    while (xp >= experienceService.requiredForLevel(level)) {
      xp -= experienceService.requiredForLevel(level);
      level++;
    }
    return xp;
  }

  Future<Car> discoverFromPhoto(String imagePath) async {
    final result = await recognition.recognize(imagePath);
    final plate = await plateDetection.detect(imagePath);
    final safePhoto = await privacy.maskPlates(imagePath, plate);

    final rarity = rarityService.calculate(
      brand: result.brand,
      model: result.model,
      color: result.color,
    );

    final base = rarityService.baseValueFor(result.brand, result.model);
    final id = '${DateTime.now().microsecondsSinceEpoch}_${Random().nextInt(99999)}';

    final car = Car(
      id: id,
      brand: result.brand,
      model: result.model,
      generation: result.generation,
      bodyType: result.bodyType,
      color: result.color,
      rarity: rarity,
      baseValue: base,
      currentValue: base,
      speed: 50 + rarity.tier * 8,
      handling: 45 + rarity.tier * 9,
      durability: 45 + rarity.tier * 9,
      maxSpeed: 80 + rarity.tier * 8,
      maxHandling: 80 + rarity.tier * 7,
      maxDurability: 80 + rarity.tier * 7,
      level: 1,
      experience: 0,
      damage: 0,
      photoPath: safePhoto,
      discoveredAt: DateTime.now(),
      status: CarStatus.outside,
    );

    state.cars.add(car);
    state.money += economyService.discoveryReward(rarity);
    state.totalXp += experienceService.xpForDiscovery(rarity);

    await store.save();
    notifyListeners();
    return car;
  }

  Future<void> addDemoCar() async {
    await discoverFromPhoto('mock://camera/${DateTime.now().millisecondsSinceEpoch}.jpg');
  }

  Future<void> putInGarage(Car car) async {
    garageService.store(car, state.cars);
    await store.save();
    notifyListeners();
  }

  Future<void> removeFromGarage(Car car) async {
    garageService.remove(car);
    await store.save();
    notifyListeners();
  }

  Future<bool> buyGarageSlot() async {
    final nextSlot = state.garageSlots + 1;
    final price = economyService.garagePrice(nextSlot);
    if (state.money < price) return false;
    state.money -= price;
    state.garageSlots = nextSlot;
    garageService.slots = nextSlot;
    await store.save();
    notifyListeners();
    return true;
  }

  Future<bool> upgrade(Car car, UpgradeType type) async {
    final cost = economyService.upgradeCost(car, type);
    if (state.money < cost) return false;

    try {
      upgradeService.upgrade(car, type);
    } catch (_) {
      return false;
    }

    state.money -= cost;
    await store.save();
    notifyListeners();
    return true;
  }

  Future<bool> repair(Car car) async {
    final cost = car.repairCost;
    if (cost <= 0) return true;
    if (state.money < cost) return false;
    state.money -= cost;
    upgradeService.repair(car);
    await store.save();
    notifyListeners();
    return true;
  }

  Future<bool> sell(Car car) async {
    if (car.status == CarStatus.garaged) return false;
    final price = marketService.price(car);
    state.money += price;
    state.cars.removeWhere((c) => c.id == car.id);
    await store.save();
    notifyListeners();
    return true;
  }

  Future<void> startRace(Car car, {required bool stake}) async {
    if (!car.isUsable) throw StateError('Эта машина сейчас недоступна');

    final random = Random();
    final playerScore = car.speed * 0.45 + car.handling * 0.35 + car.durability * 0.20;
    final botScore = 55 + random.nextInt(45);
    final win = playerScore + random.nextDouble() * 20 >= botScore;

    upgradeService.applyDamage(car, 5 + random.nextInt(21));

    state.totalXp += experienceService.xpForRace(win: win);
    state.money += economyService.raceReward(win: win);

    await store.save();
    notifyListeners();
  }

  Future<void> startChase(Car car) async {
    car.status = CarStatus.inChase;
    notifyListeners();

    await Future<void>.delayed(const Duration(seconds: 3));

    final success = Random().nextDouble() > 0.22;
    if (success) {
      car.status = car.garageSlot != null ? CarStatus.garaged : CarStatus.recovered;
      state.totalXp += experienceService.xpForChase();
      state.money += economyService.chaseReward();
    } else {
      // Защита от безвозвратной потери:
      // автомобиль остаётся доступным для повторной попытки.
      car.status = car.garageSlot != null ? CarStatus.garaged : CarStatus.outside;
      state.totalXp += 20;
    }

    await store.save();
    notifyListeners();
  }

  Future<void> simulateTheft(Car car) async {
    if (car.isInGarage || car.status == CarStatus.stolen) return;
    car.status = CarStatus.stolen;
    await store.save();
    notifyListeners();
  }

  Future<void> recoverImmediately(Car car) async {
    // MVP-safe recovery: предотвращает permanent loss.
    car.status = car.garageSlot != null ? CarStatus.garaged : CarStatus.recovered;
    await store.save();
    notifyListeners();
  }

  Future<void> connectBluetooth() async {
    await networkService.discoverNearby();
    await networkService.connect('mock-peer');
    notifyListeners();
  }
}

class RaceScreen extends StatefulWidget {
  final GameController controller;
  final Car car;

  const RaceScreen({super.key, required this.controller, required this.car});

  @override
  State<RaceScreen> createState() => _RaceScreenState();
}

class _RaceScreenState extends State<RaceScreen> {
  double playerX = 0.5;
  double botX = 0.5;
  double progress = 0;
  bool finished = false;
  bool? win;
  Timer? timer;

  @override
  void initState() {
    super.initState();
    timer = Timer.periodic(const Duration(milliseconds: 80), (_) {
      if (finished) return;
      setState(() {
        progress += 0.012 + widget.car.speed / 20000;
        playerX = (playerX + (Random().nextDouble() - 0.5) * 0.08).clamp(0.12, 0.88);
        botX = (botX + (Random().nextDouble() - 0.5) * 0.06).clamp(0.12, 0.88);
        if (progress >= 1) {
          finished = true;
          win = widget.car.speed + widget.car.handling >= 125 || Random().nextDouble() > 0.45;
        }
      });

      if (finished) {
        _finish();
      }
    });
  }

  Future<void> _finish() async {
    timer?.cancel();
    await widget.controller.startRace(widget.car, stake: false);
  }

  @override
  void dispose() {
    timer?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('ГОНКА')),
      body: Column(
        children: [
          Expanded(
            child: CustomPaint(
              painter: RacePainter(
                playerX: playerX,
                botX: botX,
                progress: progress,
              ),
              child: const SizedBox.expand(),
            ),
          ),
          Padding(
            padding: const EdgeInsets.all(16),
            child: finished
                ? Column(
                    children: [
                      Text(
                        win == true ? 'ПОБЕДА!' : 'ПОРАЖЕНИЕ',
                        style: TextStyle(
                          fontSize: 34,
                          fontWeight: FontWeight.w900,

class ChaseScreen extends StatefulWidget {
  final GameController controller;
  final Car car;

  const ChaseScreen({super.key, required this.controller, required this.car});

  @override
  State<ChaseScreen> createState() => _ChaseScreenState();
}

class _ChaseScreenState extends State<ChaseScreen> {
  int seconds = 600;
  Timer? timer;
  bool ended = false;

  @override
  void initState() {
    super.initState();
    timer = Timer.periodic(const Duration(seconds: 1), (_) {
      if (seconds <= 0) {
        timer?.cancel();
        return;
      }
      setState(() => seconds--);
    });
  }

  @override
  void dispose() {
    timer?.cancel();
    super.dispose();
  }

  Future<void> _catch() async {
    if (ended) return;
    ended = true;
    timer?.cancel();
    await widget.controller.startChase(widget.car);
    if (!mounted) return;
    showDialog(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('Погоня завершена'),
        content: Text(
          widget.car.status == CarStatus.inChase
              ? 'Ошибка состояния'
              : 'Автомобиль обработан системой погони.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.popUntil(context, (r) => r.isFirst),
            child: const Text('ОК'),
          ),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final mins = seconds ~/ 60;
    final secs = seconds % 60;

    return Scaffold(
      appBar: AppBar(title: const Text('ПОГОНЯ')),
      body: Column(
        children: [
          Expanded(
            child: CustomPaint(
              painter: ChasePainter(),
              child: const SizedBox.expand(),
            ),
          ),
          Padding(
            padding: const EdgeInsets.all(16),
            child: Column(
              children: [
                Text(
                  '$mins:${secs.toString().padLeft(2, '0')}',
                  style: const TextStyle(fontSize: 30, fontWeight: FontWeight.w900),
                ),
                const Text('Окно погони — 10 минут'),
                const SizedBox(height: 10),
                FilledButton.icon(
                  onPressed: _catch,
                  icon: const Icon(Icons.local_police),
                  label: const Text('ПЕРЕХВАТИТЬ'),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
}

class BluetoothScreen extends StatefulWidget {

  const BluetoothScreen({super.key, required this.controller});

  @override
  State<BluetoothScreen> createState() => _BluetoothScreenState();
}

class _BluetoothScreenState extends State<BluetoothScreen> {
  bool scanning = false;
  bool connected = false;

  Future<void> _scan() async {
    setState(() => scanning = true);
    await widget.controller.connectBluetooth();
    if (!mounted) return;
    setState(() {
      scanning = false;
      connected = true;
    });
  }

  @override
  Widget build(BuildContext context) {
    final cars = widget.controller.state.cars;

    return Scaffold(
      appBar: AppBar(title: const Text('Bluetooth')),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          ListTile(
            leading: Icon(
              connected ? Icons.bluetooth_connected : Icons.bluetooth_disabled,
            ),
            title: Text(connected ? 'Игрок подключён' : 'Нет подключения'),
            subtitle: const Text('В production здесь будет реальный BLE/P2P слой'),
          ),
          FilledButton.icon(
            onPressed: scanning ? null : _scan,
            icon: const Icon(Icons.search),
            label: Text(scanning ? 'Поиск...' : 'Найти игроков'),
          ),
          const SizedBox(height: 18),
          const Text('Обмен карточками', style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
          const SizedBox(height: 8),
          if (cars.isEmpty)
            const Text('Нет карточек для обмена')
          else
            ...cars.take(5).map(
              (car) => Card(
                child: ListTile(
                  leading: CarIcon(car: car),
                  title: Text(car.title),
                  subtitle: Text(car.rarity.label),
                  trailing: connected
                      ? OutlinedButton(
                          onPressed: () => _tradeMock(context, car),
                          child: const Text('Предложить'),
                        )
                      : null,
                ),
              ),
            ),
          const SizedBox(height: 16),
          const Text(
            'Протокол сделки: transaction ID → предложение → '
            'подтверждение A → подтверждение B → атомарная передача. '
            'Повторный transaction ID должен быть отклонён.',
            style: TextStyle(color: Colors.white70),
          ),
        ],
      ),
    );
  }

  void _tradeMock(BuildContext context, Car car) {
    final tx = 'tx_${DateTime.now().microsecondsSinceEpoch}';
    final ok = widget.controller.tradeService.begin(tx, car.id, 'remote-card');
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(ok
            ? 'Сделка $tx создана. Требуются подтверждения обеих сторон.'
            : 'Карточка уже участвует в сделке.'),
      ),
    );
  }
}

class RacePainter extends CustomPainter {
  final double playerX;
  final double botX;
  final double progress;

  RacePainter({
    required this.playerX,
    required this.botX,
    required this.progress,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final road = Paint()..color = const Color(0xFF20242A);
    canvas.drawRect(
      Rect.fromLTWH(size.width * .15, 0, size.width * .70, size.height),
      road,
    );

    final lane = Paint()
      ..color = Colors.white24
      ..strokeWidth = 3;

    for (double y = -30; y < size.height; y += 50) {
      final offset = (progress * 150) % 50;
      canvas.drawLine(
        Offset(size.width * .50, y + offset),
        Offset(size.width * .50, y + offset + 24),
        lane,
      );
    }

    _drawCar(
      canvas,
      Offset(size.width * playerX, size.height * .78),
      Colors.orange,
    );
    _drawCar(
      canvas,
      Offset(size.width * botX, size.height * .25),
      Colors.blue,
    );
  }

  void _drawCar(Canvas canvas, Offset center, Color color) {
    final p = Paint()..color = color;
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromCenter(center: center, width: 34, height: 62),
        const Radius.circular(9),
      ),
      p,
    );
  }

  @override
  bool shouldRepaint(covariant RacePainter oldDelegate) => true;
}

class ChasePainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final road = Paint()..color = const Color(0xFF20242A);
    canvas.drawRect(
      Rect.fromLTWH(size.width * .12, 0, size.width * .76, size.height),
      road,
    );

    final side = Paint()..color = Colors.green.withOpacity(.18);
    canvas.drawRect(Rect.fromLTWH(0, 0, size.width * .12, size.height), side);
    canvas.drawRect(
      Rect.fromLTWH(size.width * .88, 0, size.width * .12, size.height),
      side,
    );

    final lane = Paint()
      ..color = Colors.white24
      ..strokeWidth = 3;

    for (double y = 0; y < size.height; y += 50) {
      canvas.drawLine(
        Offset(size.width * .50, y),
        Offset(size.width * .50, y + 24),
        lane,
      );
    }

    final police = Paint()..color = Colors.orange;
    final criminal = Paint()..color = Colors.red;

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromCenter(
          center: Offset(size.width * .50, size.height * .72),
          width: 38,
          height: 70,
        ),
        const Radius.circular(10),
      ),
      police,
    );

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromCenter(
          center: Offset(size.width * .50, size.height * .28),
          width: 38,
          height: 70,
        ),
        const Radius.circular(10),
      ),
      criminal,
    );
  }

  @override
  bool shouldRepaint(covariant ChasePainter oldDelegate) => false;
}
